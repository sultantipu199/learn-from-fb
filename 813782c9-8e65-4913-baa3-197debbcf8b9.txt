function createGoogleDoc() {
  const sheet = SpreadsheetApp.getActiveSheet();
  const data = sheet.getDataRange().getValues();
  
  // নতুন Google Doc তৈরি করুন
  const doc = DocumentApp.create("Facebook Video Lessons");
  const body = doc.getBody();
  
  let lessonNo = 1;
  
  // প্রথম রো স্কিপ করুন (হেডার)
  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    
    // Lesson টাইটেল যোগ করুন
    body.appendParagraph("LESSON " + lessonNo).setHeading(DocumentApp.ParagraphHeading.HEADING1);
    
    // ভিডিও তথ্য যোগ করুন
    body.appendParagraph("Video Title: " + row[1]);
    body.appendParagraph("Video URL: " + row[2]);
    body.appendParagraph("Posted Date: " + row[3]);
    body.appendParagraph("Topic: " + row[4]);
    body.appendParagraph("Key Points: " + row[5]);
    
    // ফাঁকা লাইন
    body.appendParagraph("═════════════════════════════════════════");
    body.appendParagraph("");
    
    lessonNo++;
  }
  
  Logger.log("Google Doc তৈরি হয়েছে: " + doc.getUrl());
}
```
